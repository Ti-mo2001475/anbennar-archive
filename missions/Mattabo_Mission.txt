	F70_contact_with_the_harpies = {
		position = 4
		required_missions = {
			F70_repel_the_invaders
		}
		#The harpies of Duwarkani (F62) can be dealt with in three ways:
		#  1. secure their friendship (they are independent, guaranteed and trusting us, while we are stronger)
		#  2. infiltrate whoever owns them with a spy network
		#  3. simply own them (annexed, or held by one of our subjects)
		trigger = {
			OR = {
				#1. Secured good relations with the harpies
				AND = {
					exists = F62 #Duwarkani
					F62 = {
						is_subject = no #Independent
						guaranteed_by = ROOT
						trust = {
							who = ROOT
							value = 60
						}
					}
					army_size = F62 #Our army is larger than Duwarkani's
				}
				#2. Deep infiltration of the harpy realm (or of its overlord)
				F62 = {
					tag_or_overlord_has_spy_network_from = {
						who = ROOT #Our spy network in Duwarkani or its overlord
						value = 25
					}
				}		
				#3. Duwarkani is ours (annexed, or held by our subject)
				6022 = { #Duwarkani
					OR = {
						owned_by = ROOT
						owned_by_subject_of = ROOT
					}
				}
			}
		}
		effect = {
			if = {
				#Friendship with the harpies was secured
				limit = {
					exists = F62 #Duwarkani
					F62 = {
						is_subject = no #Independent
						guaranteed_by = ROOT
						trust = {
							who = ROOT
							value = 60
						}
					}
					army_size = F62 #Our army is larger than Duwarkani's
				}
				custom_tooltip = F70_contact_with_the_harpies_yes_tt
				F62 = {
					country_event = {
						id = flavour_mattabo.11 #Mattabo Promise to Protect
					}
				}
				#Friendship was not secured: preview the other outcome
				custom_tooltip = F70_contact_with_the_harpies_otherwise_tt
				tooltip = {
					duwarkani_area = {
						add_permanent_claim = ROOT
					}
				}
				custom_tooltip = F70_contact_with_the_harpies_rebellious_harpies_tt
			}
			else = {
				#The harpies are brought to heel
				custom_tooltip = F70_contact_with_the_harpies_no_tt
				tooltip = {
					F62 = {
						country_event = {
							id = flavour_mattabo.11
						}
					}
				}
				custom_tooltip = F70_contact_with_the_harpies_otherwise_tt
				duwarkani_area = {
					add_permanent_claim = ROOT
				}
				custom_tooltip = F70_contact_with_the_harpies_rebellious_harpies_tt
				hidden_effect = {
					duwarkani_area = {
						limit = {
							NOT = { owned_by = ROOT }
							NOT = { owned_by_subject_of = ROOT }
						}
						add_province_modifier = {
							name = F70_rebellious_harpies
							duration = 7300 #20 years
						}
					}
				}
			}
		}
	}

	F70_reconcile_with_brothers = {
		position = 11
		required_missions = {
			F70_gnomish_equipment
		}
		#The source screenshot ("Reconcile with Brothers") lists four hard
		#requirements plus one "either/or" pair, and three effects. The Ahati are
		#the two brotherhoods - the Northern one seated in Aakheta (466) and the
		#Southern one in Ka-Obix (483) - so this mission is about mending the rift
		#with them after the gnollish wars and giving the Northern Brotherhood the
		#upper hand.
		trigger = {
			#Requirement 1: the Ahati estate is at 60 loyalty or more
			estate_loyalty = {
				estate = estate_ahati
				loyalty = 60
			}
			#Requirement 2: the Ahati hold the [Grant Noble Titles] privilege
			#(`estate_ahati_noble_titles` in
			#`common/estate_privileges/estate_ahati_privileges.txt`)
			has_estate_privilege = estate_ahati_noble_titles
			#Requirement 3: the Ahati hold the [Encourage Militarization]
			#privilege (`estate_ahati_militarization`, same file)
			has_estate_privilege = estate_ahati_militarization
			#Requirement 4: the Steward reform [Bronze Dragon Academy] is enacted
			#(`build_academy_reform` in `common/imperial_reforms/01_china.txt`).
			#`empire_of_china_reform_passed` is a trigger, so it can only ever be
			#checked here - never written into an `effect` block.
			empire_of_china_reform_passed = build_academy_reform
			#Requirement 5: ONE of the two following branches
			#  1. the Ahati have completed at least 3 of their own estate agendas
			#  2. the Steward reform [Lessons of Dameria] is enacted
			#The screenshot spells the second reform "Lessons of Dameira"; the
			#reform in the repository is `lessons_dameria_reform` (display name
			#"Lessons of Dameria"). It is the reform that unlocks the
			#`khet_mandate_complete_ahati_agenda` decision ("Ahati: Leverage
			#Authority", which auto-completes an Ahati agenda), so both branches
			#amount to "we no longer depend on the agendas themselves".
			OR = {
				num_of_estate_agendas_completed = {
					estate = estate_ahati
					value = 3
				}
				empire_of_china_reform_passed = lessons_dameria_reform
			}
		}
		effect = {
			#Ahati Loyalty +10%
			add_estate_loyalty = {
				estate = estate_ahati
				loyalty = 10
			}
			#Enable the Ahati privilege [Support Northern Brotherhood]
			#(`estate_ahati_support_north`). Note that the privilege's own
			#`is_valid` / `can_select` demand that we own the core province
			#Aakheta (466); without it the privilege is judged invalid and is
			#revoked again immediately, so it is worth taking Aakheta first.
			set_estate_privilege = estate_ahati_support_north
			#Gain the country modifier [Brothers of One Mind?] for 25 years
			#(`F70_brothers_of_one_mind_but_actually_not` in
			#`common/event_modifiers/anb_Mattabo_modifiers.txt`, whose keys are
			#exactly Ahati Loyalty Equilibrium +10% and Shock Damage +5%). No
			#`custom_tooltip` here: per the author's 2026-10-08 rule a missing
			#display name loc key is never a reason to add one, and
			#`add_country_modifier` brings its own tooltip line.
			add_country_modifier = {
				name = F70_brothers_of_one_mind_but_actually_not
				duration = 9125
			}
		}
	}

		position = 13
		required_missions = {
			F70_reconcile_with_brothers
		}
		trigger = {
			#The Ahati estate has the [Ahati Commanders] privilege
			#(`estate_ahati_ahati_commanders`, defined in
			#`common/estate_privileges/estate_ahati_privileges.txt`; the mission
			#`F70_the_guardians` grants it, so it is normally already held here)
			has_estate_privilege = estate_ahati_ahati_commanders
			#A Commandant advisor of at least skill 4, or any military advisor of
			#skill 5, is employed. Advisor levels 4 and 5 are both a Cradle of
			#Civilization feature - without it advisor skill caps at 3 - so this
			#whole requirement is DLC-gated, and the third branch spells that out
			#with `has_dlc`, exactly like `has_mil_advisor_5` in
			#common/scripted_triggers/anb_scripted_triggers.txt:2228 does. (The
			#screenshot has no "or a level 3 advisor" fallback, unlike
			#`F70_gnomish_equipment`, so none is added here.)
			#(`employed_advisor`'s `skill` is an exact value, not a floor, so
			#"at least 4" has to be spelled out per skill level.)
			OR = {
				commandant = 4
				commandant = 5
				AND = {
					has_dlc = "Cradle of Civilization"
					has_mil_advisor_5 = yes
				}
			}
			#Aakheta (466) has been developed and built up
			466 = { #Aakheta
				development = 30
				num_of_buildings_in_province = 4
			}
			#The Bronze Dragon Academy has reached at least the "Significant"
			#level. A great project at tier 0 is Insignificant, 1 Noteworthy,
			#2 Significant and 3 Monumental, and `tier = N` reads "at least N"
			#(same read as this tree's `F70_targeted_training`, which asks for
			#Tower of Muhaqaar at `tier = 2` = Significant). `tier = 2` is what
			#matches the screenshot's "at least Significant"; `tier = 3` is the
			#fully built academy and is a separate, later requirement.
			466 = {
				has_great_project = {
					type = bronze_dragon_academy
					tier = 2
				}
			}
		}
		effect = {
			#The [Ahati Commanders] privilege is improved: Land Leader Manuever
			#+1, Land Leader Siege +1, and (Shadow Warriors only) Discipline +5%.
			#TODO: the whole effect is left unimplemented for now. Adding the

			#      the rule is "mission tree file only" (author decision
			#      2026-10-08, recorded in .memory/mtb_mission/mt_progress.md
			#      and .generic/anb_script_file_conventions.md), so the earlier
			#      attempt - two `conditional_modifier` blocks in
			#      `estate_ahati_ahati_commanders`
			#      (common/estate_privileges/estate_ahati_privileges.txt) with
			#      `trigger = { mission_completed = F70_learn_from_brothers }`,
			#      one for `leader_land_manuever = 1` + `leader_siege = 1` and a
			#      separate one for `discipline = 0.05` +
			#      `is_mamluks_modifier = yes` - has been withdrawn and the

			#      When the tree is finished, re-apply that pair of blocks. Two
			#      notes for then: they must stay two separate
			#      `conditional_modifier` blocks, because
			#      `is_mamluks_modifier = yes` scopes an entire block to the
			#      Shadow Warriors and sharing one block would restrict the
			#      leader pips to them as well; and the key spellings are
			#      `leader_land_manuever` (the engine's spelling, not
			#      `leader_land_maneuver`) and `leader_siege` (there is no

			#      `benefits` instead - that would hand the bonuses to every

			#      Nothing is applied in this effect, which also means no

			#      reason).
		}
	}

	F70_take_the_initiative = {
		position = 11
		required_missions = {
			F70_diabolical_sisters
		}
		#The gnollish packs of Gazraak and Dreksaret are the last Seascourge
		#strongholds left in the Sorrow, and their lands are already under our
		#permanent claims (see `F70_diabolical_sisters`). This mission is about
		#taking the initiative - and their people - in hand: the core land of the
		#three gnollish towns must be worked by our own officials, and the towns
		#themselves become the slave markets that feed the rest of our realm.
		trigger = {
			#Every gnollish province of the Gazraak and Dreksaret regions is our
			#core province and has left Xhazobkult and Kvangahga behind (a province
			#that still has [Religious Zeal] counts as freshly converted).
			#The non-gnollish provinces of these two regions (the Ofehibi and
			#Raghamidesh lands of the Dreksaret area) are deliberately ignored.
			#`limit` must NOT be used to do that here: inside a trigger, a multi
			#object scope (region / area / province group) ignores a bare `limit`,
			#so the culture exemption is spelled out as the first branch of the
			#`OR` instead - `NOT = { culture_group = gnollish }` makes every
			#non-gnollish province pass on its own, while every gnollish province
			#has to satisfy the second branch.
			#Every province of both regions is settled, so no exemption for
			#uncolonized land is needed here.
			gazraak_region = {
				type = all
				OR = {
					NOT = { culture_group = gnollish }
					AND = {
						owned_by = ROOT
						is_core = ROOT
						OR = {
							has_province_modifier = religious_zeal_at_conv
							AND = {
								NOT = { religion = xhazobkult }
								NOT = { religion = kvangahga }
							}
						}
					}
				}
			}
			dreksaret_region = {
				type = all
				OR = {
					NOT = { culture_group = gnollish }
					AND = {
						owned_by = ROOT
						is_core = ROOT
						OR = {
							has_province_modifier = religious_zeal_at_conv
							AND = {
								NOT = { religion = xhazobkult }
								NOT = { religion = kvangahga }
							}
						}
					}
				}
			}

				OR = {
					has_production_building_trigger = yes
					has_trade_building_trigger = yes
				}
				num_of_units_in_province = {
					who = ROOT
					amount = 1

			}

	F70_ensure_enchanted_steel_supply = {
		position = 12
		required_missions = {
			F70_take_the_initiative
		}
		trigger = {
			#Kvangraak itself has grown into a forge city: a base production of
			#at least 10, a manufactory, and 10k of our soldiers stationed
			#there (1 unit = 1k). `has_manufactory_trigger` is the repo's
			#standard "has a manufactory" check.
			5442 = { #Kvangraak
				owned_by = ROOT
				base_production = 10
				has_manufactory_trigger = yes
				num_of_units_in_province = {
					who = ROOT
					amount = 10
				}
			}
			#Its infrastructure has been expanded at least once. Without the
			#Leviathan DLC that action does not exist at all, so the reference
			#mission asks for 50 adm power instead - the effect below spends it.
			if = {
				limit = { has_dlc = "Leviathan" }
				5442 = { num_of_times_expanded_infrastructure = 1 }
			}
			else = { adm_power = 50 }
			#Every province of the Kvangraak area is ours and has both a
			#workshop/counting house and barracks/training fields. `type = all`
			#makes this an "every province" check (an area scope with plain
			#province triggers is ambiguous).
			kvangraak_area = {
				type = all
				owned_by = ROOT
				has_production_building_trigger = yes
				has_manpower_building_trigger = yes
			}
		}
		effect = {
			#The two cost lines of the reference screenshot ("Lose 10k armies
			#in Kvangraak" and "Destroys the great project Esuvrem") are
			#effects, so they are implemented with `kill_units` and
			#`destroy_great_project` (both of them are province scoped).
			5442 = { #Kvangraak
				kill_units = {
					who = ROOT
					amount = 10
				}
				destroy_great_project = { type = esuvrem }
			}
			#Permanent country modifier [Enchanted Steel Equipment]:
			#(Only to Shadow Warriors:) Fire Damage Received -10% and
			#Shock Damage Received -10%. It has no display name loc key yet;
			#per the 2026-10-08 decision no `custom_tooltip` is added for it,
			#so this line shows the raw id until the key exists.
			add_country_modifier = {
				name = F70_enchanted_steel_equipment
				duration = -1
			}
			#Without the Leviathan DLC the 50 adm power that stand in for the
			#infrastructure requirement are paid here.
			if = {
				limit = { NOT = { has_dlc = "Leviathan" } }
				add_adm_power = -50
			}
		}
