#Kaino Koalition Fires
country_event = {
	id = flavour_kaino.7
	title = flavour_kaino.7.t
	desc = flavour_kaino.7.d
	picture = MOSSI_CAVALRY_eventPicture
	
	is_triggered_only = yes

	immediate = {
		every_known_country = {
			limit = { 
				capital_scope = {
					superregion = fangaula_superregion
				}
				NOT = {
					is_subject_of = ROOT
				}
			}
			set_country_flag = kaino_koalition_member
		}
	}
	option = {
		name = flavour_kaino.7.a
		custom_tooltip = kaino_koalition_will_fire_tt
		add_country_modifier = {
			name = kaino_koalition_temp_bonus
			duration = 1825
		}
		hidden_effect = {
			add_ae_for_known_countries_with_limit = {
				limit = "
					has_country_flag = kaino_koalition_member
					NOT = { is_subject = yes }
				"
				value = 60
				apply_calc = no
				effect = "
					set_ai_attitude = { attitude = attitude_outraged who = ROOT }
					form_coalition_against = ROOT
				"
			}
			random_known_country = {
				limit = { 
					capital_scope = {
						superregion = fangaula_superregion
					}
					has_country_flag = kaino_koalition_member
					coalition_target = ROOT
				}
				declare_war_with_cb = {
					casus_belli = cb_super_badboy
					who = ROOT
				}
			}
		}
	}
	after = {
		hidden_effect = {
			set_country_flag = kaino_koalition_fired
		}	
	}
}
