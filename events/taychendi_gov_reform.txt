namespace = taychendi_gov_reform

# Yearly glory from subjects
country_event = {
    id = taychendi_gov_reform.1
    title = taychendi_gov_reform.1.t
    desc = taychendi_gov_reform.1.d
    picture = CITY_eventPicture

    is_triggered_only = yes
    hidden = yes          

    option = {
        name = taychendi_gov_reform.1.a
        if = {
            limit = { num_of_subjects = 5 }
            gain_taychend_glory_effect = { amount = 25 }
        }
        else_if = {
            limit = { num_of_subjects = 4 }
            gain_taychend_glory_effect = { amount = 20 }
        }
        else_if = {
            limit = { num_of_subjects = 3 }
            gain_taychend_glory_effect = { amount = 15 }
        }
        else_if = {
            limit = { num_of_subjects = 2 }
            gain_taychend_glory_effect = { amount = 10 }
        }
        else_if = {
            limit = { num_of_subjects = 1 }
            gain_taychend_glory_effect = { amount = 5 }
        }
    }
}
