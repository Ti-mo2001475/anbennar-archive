# Olavish Dominance Upgrade

	id = olavish_minor.1
	title = olavish_minor.1.t
	desc = olavish_minor.1.d
	picture = CARIBBEAN_PIRATE_FORT_eventPicture

	is_triggered_only = yes

	trigger = {
		always = yes
	}

	option = {
		name = olavish_minor.1.a
		ai_chance = {
			factor = 100
		}
		if = {
			limit = {
				has_country_modifier = olavish_minor_olavish_dominance_1
			}
			remove_country_modifier = olavish_minor_olavish_dominance_1
			add_country_modifier = {
				name = olavish_minor_olavish_dominance_2
				duration = -1
			}
		}
		else_if = {
			limit = {
				has_country_modifier = olavish_minor_olavish_dominance_2
			}
			remove_country_modifier = olavish_minor_olavish_dominance_2
			add_country_modifier = {
				name = olavish_minor_olavish_dominance_3
				duration = -1
			}
		}
	}
}
