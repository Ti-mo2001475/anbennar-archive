country_event = { #Quest for Reehanviza End
	id = flavour_banbe_anen.1000
	title = flavour_banbe_anen.1000.t
	desc = flavour_banbe_anen.1000.d
	picture = MILITARY_CAMP_eventPicture

	is_triggered_only = yes
	trigger = { }

	option = {
		name = flavour_banbe_anen.1000.a
		trigger = {
			OR = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = order }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = chaos }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = household_gods }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = outer_realms }
			}
		}

		if = { #Order Mastered
			limit = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = order }
			}
			add_ruler_modifier = {
				name = banbe_anen_quest_for_Reehanviza_mastered_order
				duration = -1
			}
		}
		if = { #Chaos Mastered
			limit = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = chaos }
			}
			add_ruler_modifier = {
				name = banbe_anen_quest_for_Reehanviza_mastered_chaos
				duration = -1
			}
		}
		if = { #Household Gods Mastered
			limit = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = household_gods }
			}
			add_ruler_modifier = {
				name = banbe_anen_quest_for_Reehanviza_mastered_household_gods
				duration = -1
			}
		}
		if = { #Outer Realms Mastered
			limit = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = outer_realms }
			}
			add_ruler_modifier = {
				name = banbe_anen_quest_for_Reehanviza_mastered_outer_realms
				duration = -1
			}
		}
		remove_country_modifier = banbe_anen_quest_for_Reehanviza
	}

	option = {
		name = flavour_banbe_anen.1000.b
		trigger = {
			NOT = {
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = order }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = chaos }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = household_gods }
				banbe_anen_quest_for_Reehanviza_mastery_trigger = { which = outer_realms }
			}
		}

		add_prestige = -50
		add_legitimacy_equivalent = { amount = -50 republican_tradition = -30 }
		add_stability = -1
		remove_country_modifier = banbe_anen_quest_for_Reehanviza
	}
}
