#A Newborn Island
country_event = {
	id = flavour_volcanic_truth.1
	title = flavour_volcanic_truth.1.t
	desc = flavour_volcanic_truth.1.d
	picture = FAR_AND_WIDE_eventPicture	
	goto = newborn_island_province
	
	is_triggered_only = yes	
	trigger = {
		religion = volcanic_truth
		any_owned_province = {
			has_port = yes
			superregion = kanolowele_superregion
		}
	}
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = kanolowele_superregion
					has_port = yes
				}
				save_event_target_as = newborn_island_province
			}
		}
	}
	
	#Proclaim a month of celebration
	option = {
		name = flavour_volcanic_truth.1.a
		add_stability = 1
	}
	
	#Dedicate this island to Quahopilo
	option = {
		name = flavour_volcanic_truth.1.b
		add_country_modifier = {
			name = island_dedication_quahopilo
			duration = 5475
		}
	}
	
	#Dedicate the island to Muatlaqua
	option = {
		name = flavour_volcanic_truth.1.c
		add_country_modifier = {
			name = island_dedication_muatlaqua
			duration = 5475
		}
	}
	
	#Profit from the mass of pilgrims
	option = {
		name = flavour_volcanic_truth.1.e
		if = {
			limit = { monthly_income = 21 }
			add_treasury = 250
        }
        else = { add_years_of_income = 1 }
	}
}

#Visions of Glory
country_event = {
	id = flavour_volcanic_truth.2
	title = flavour_volcanic_truth.2.t
	desc = flavour_volcanic_truth.2.d
	picture = PIETY_eventPicture	

	is_triggered_only = yes	
	trigger = {
		religion = volcanic_truth
		any_owned_province = {
			superregion = kanolowele_superregion
		}
		has_heir = yes
		NOT = { heir_is_powerful_mage = yes }
		has_estate = estate_mages
	}
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = kanolowele_superregion
				}
				save_event_target_as = visions_of_glory_province
			}
		}
	}
	
	#Allow it
	option = {
		name = flavour_volcanic_truth.2.a
		random_list = {
			90 = { kill_heir = {} }
			10 = { define_powerful_mage = { type = heir } }
		}
	}
	
	#Forbid it
	option = {
		name = flavour_volcanic_truth.2.b
		add_estate_loyalty = {
			estate = estate_church
			loyalty = -5
		}
	}
	
	#Allow it, with suitable arcane safeguards
	option = {
		name = flavour_volcanic_truth.2.c
		highlight = yes
		#Trigger: Proficient in Abjuration, not legendary in abjuration
		trigger = { 
			is_overall_abjuration_level_1_minimum = yes 
			NOT = { is_overall_abjuration_level_3 = yes }
		}
		add_country_modifier = {
			name = maintaining_volcanic_spells
			duration = 3650
		}
		random_list = {
			50 = { kill_heir = {} }
			50 = { define_powerful_mage = { type = heir } }
		}
	}
	
	#Allow it, with flawless arcane safeguards
	option = {
		name = flavour_volcanic_truth.2.e
		highlight = yes
		#trigger: legendary in abjuration
		trigger = { 
			is_overall_abjuration_level_3 = yes 
		}
		add_country_modifier = {
			name = maintaining_volcanic_spells
			duration = 5475
		}
		define_powerful_mage = { type = heir }
	}
	
	#Commission the artificers to create suitable protection
	option = {
		name = flavour_volcanic_truth.2.f
		highlight = yes
		trigger = {
			has_estate = estate_artificers
		}
		add_country_modifier = {
			name = maintaining_volcanic_artifice
			duration = 1825
		}
		random_list = {
			50 = { kill_heir = {} }
			50 = { define_powerful_mage = { type = heir } }
		}
	}
	
	#The Clergy have rituals that may help, but they are not cheap
	option = {
		name = flavour_volcanic_truth.2.g
		change_estate_land_share = { 
			estate = estate_church 
			share = 5 
		}
		random_list = {
			67 = { kill_heir = {} }
			33 = { define_powerful_mage = { type = heir } }
		}
	}
}


#A Gift From the Volcano?
country_event = {
	id = flavour_volcanic_truth.3
	title = flavour_volcanic_truth.3.t
	desc = flavour_volcanic_truth.3.d
	picture = DAMESTEAR_CORE_eventPicture

	is_triggered_only = yes	
	trigger = {
		religion = volcanic_truth
		any_owned_province = {
			superregion = kanolowele_superregion
		}
	}
	
	goto = volcano_reward_province
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					superregion = kanolowele_superregion
				}
				save_event_target_as = volcano_reward_province
			}
		}
	}
	
	#We cannot risk it
	option = {
		name = flavour_volcanic_truth.3.a
		add_estate_loyalty = {
			estate = estate_church
			loyalty = -10
		}
	}
	
	#Retrieve it with sorcerous means
	option = {
		name = flavour_volcanic_truth.3.b
		trigger = {
			OR = {
				has_estate = estate_mages
				ruler_is_powerful_mage = yes				
			}
			has_mana = { amount = 25 }
		}
		change_mana = { amount = -25 }
		country_event = { 
			id = flavour_volcanic_truth.4
			days = 1
		}
	}
	
	#Have the artificers rig up a contraption to retrieve the rock
	option = {
		name = flavour_volcanic_truth.3.c
		trigger = {
			has_estate = estate_artificers
		}
		change_estate_land_share = {
			estate = estate_artificers
			share = 1
		}
		country_event = { 
			id = flavour_volcanic_truth.4
			days = 1
		}
	}
	
	#They will not like it, but task the army to find a way to haul it in.
	option = {
		name = flavour_volcanic_truth.3.e
		add_yearly_manpower = -0.2

		country_event = { 
			id = flavour_volcanic_truth.4
			days = 1
		}
	}
	
	#Hire some daring adventurers to retrieve it.
	option = {
		name = flavour_volcanic_truth.3.f
		trigger = {
			has_estate = estate_adventurers
		}
		add_years_of_income = -0.25
		country_event = { 
			id = flavour_volcanic_truth.4
			days = 1
		}
	}
}
