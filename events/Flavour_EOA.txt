		hre_size = 1 # EoA still exists
	}

		hidden_effect = { # Find the relevant elector again
			random_elector = {
				limit = { has_country_flag = eoa_protector_elector }
				save_event_target_as = expelled_elector
			}
			if = {
				limit = { NOT = { has_saved_event_target = expelled_elector } }
				random_country = {
					limit = { has_country_flag = eoa_protector_elector }
					save_event_target_as = expelled_elector
				}
			}
		}
	}

	option = {
		name = flavor_empire.44.a
		if = {  # Emperor
			limit = { is_emperor = yes }
			if = { # Increase tolerance to the race if we aren't the protector elector and if they still exist
				limit = { 
					has_saved_event_target = expelled_elector
					event_target:expelled_elector = { is_emperor = no } 
				} 
				generic_increase_of_target_primary_racial_tolerance = {
					target = event_target:expelled_elector
					size = small
				}
			}
			add_imperial_influence = 5
		}

				name = EOA_44_protector_elector

			}

		hidden_effect = {
			event_target:expelling_country = {
				eoa_end_expel_imperial_race = yes
			}
		}
	}
