#Catch off Guard
#Fired by the mission `F70_military_adventure` (missions/Mattabo_Mission.txt,
#slot 3 position 1) as the payoff of the surprise attack: the realm had been
#openly preparing for war (promote military recruitment edict in the Mattabo
#area, the ruler himself in the field, hostile trading in the Aakheta node),
#so Rakkaz is caught before it can arrange its defence. The mission's own
#trigger is unchanged - this event is only its effect.
country_event = {
	id = flavour_mattabo.1
	title = flavour_mattabo.1.t
	desc = flavour_mattabo.1.d

	is_triggered_only = yes

	trigger = {
		tag = F70
	}

	immediate = {

	}

	option = {
		name = flavour_mattabo.1.a

		# 1. Gains 5K Ahati Infantry in the capital.
		capital_scope = {
			infantry = ROOT
			infantry = ROOT
			infantry = ROOT
			infantry = ROOT
			infantry = ROOT
		}
		add_country_modifier = {
			name = F70_an_attempt
			duration = 3650
		}

		# 2. Gain permanent claims on all the Rakkaz starting provinces.
		grixzakr_area = { add_permanent_claim = ROOT }
		argazeen_area = { add_permanent_claim = ROOT }
		garpix_rakkaz_area = { add_permanent_claim = ROOT }
		5466 = { add_permanent_claim = ROOT } #Siniwix
		5467 = { add_permanent_claim = ROOT } #Siniwix
		5468 = { add_permanent_claim = ROOT } #Siniwix

		hidden_effect = {
			set_country_flag = no_callaction
			set_country_flag = F70_strike_rakkaz}
			F61 = {
				add_truce_with = ROOT
				add_trust = {
					who = ROOT
					value = -10
				}
			}
			
		}

		# The war itself. Guarded on Rakkaz still existing so that the option
		# stays safe if F71 was annexed before this event was answered;
		# `declare_war_with_cb` is the repo's standard forced-declaration effect
		# (events/Aakhetist.txt:1291, decisions/OvdalLodhumDecisions.txt:305).
		if = {
			limit = { exists = F71 }
			hidden_effect = { F71 = { set_country_flag = no_callaction } }
			declare_war_with_cb = {
				casus_belli = cb_civ_vs_monster
				who = F71
			}

		}
	}
}

#A Successful Strike
#Fired by the mission `F70_planned_victory` (missions/Mattabo_Mission.txt,
#slot 3 position 2) as the closing act of the war opened by `flavour_mattabo.1`:
#the strike succeeded, so the peace is dictated on our terms instead of being
#negotiated. The mission's own trigger is unchanged - this event is only its
#effect.
country_event = {
	id = flavour_mattabo.2
	title = flavour_mattabo.2.t
	desc = flavour_mattabo.2.d

	is_triggered_only = yes

	trigger = {
		tag = F70
	}

	immediate = {

	}

	#Single option, matching the reference screenshot's effect list:
	#  White Peace with Rakkaz
	#  Cede every province in the Argazeen, Grixzakr, Garpix Rakkaz and Siniwix
	#  areas owned by Rakkaz to Mattabo
	#  Years of Separatism in these provinces -5
	#  Cash 50
	#  Mil 100
	#  Event [Something Unexpected] in 30 days
	option = {
		name = flavour_mattabo.2.a

		if = {
			limit = {
				exists = F71
				war_with = F71
			}
			white_peace = F71
		}

		# 2-3. Rakkaz cedes every province it owns in the four areas, and each
		#      of those provinces loses 5 years of separatism.
		#      Only `owned_by = F71` is checked - the screenshot says "owned by
		#      Rakkaz", so third countries sitting in the same areas (5464 in
		#      siniwix_area starts with F55) are left alone, and these are the
		#      same four areas the mission checks.
		#      "Years of Separatism" is `add_nationalism` (the mod's own wrapper
		#      `add_years_of_separatism_in_owned_provinces` in
		#      common/scripted_effects/anb_scripted_effects.txt:14077 just calls
		#      it on every owned province). It is kept in the visible scope so it
		#      generates its own tooltip line - no `custom_tooltip` (see
		#      .generic/anb_script_file_conventions.md, section 5.1).
		argazeen_area = {
			limit = { owned_by = F71 }
			cede_province = ROOT
			add_nationalism = -5
		}
		grixzakr_area = {
			limit = { owned_by = F71 }
			cede_province = ROOT
			add_nationalism = -5
		}
		garpix_rakkaz_area = {
			limit = { owned_by = F71 }
			cede_province = ROOT
			add_nationalism = -5
		}
		siniwix_area = {
			limit = { owned_by = F71 }
			cede_province = ROOT
			add_nationalism = -5
		}

		# 4. Cash 50.
		add_treasury = 50

		# 5. Mil 100.
		add_mil_power = 100

		# 6. Event [Something Unexpected] in 30 days: the spoils of the
		#    successful strike lead straight into the next event of the starting
		#    chain, `flavour_mattabo.3` (defined right below in this file).
		country_event = {
			id = flavour_mattabo.3
			days = 30
		}

		hidden_effect = {
			clr_country_flag = no_callaction
			clr_country_flag = F70_strike_rakkaz
			if = {
				limit = { exists = F71 }
				F71 = { clr_country_flag = no_callaction }
			}
		}
	}
}

	id = flavour_mattabo.3
	title = flavour_mattabo.3.t
	desc = flavour_mattabo.3.d

	is_triggered_only = yes

	trigger = {
		tag = F70
	}

	}

		name = flavour_mattabo.3.a

		# starts, otherwise it would be dragged into it.
		grant_independence = yes

		#This Event are designed to fired in first 6 month.
		U09 = {
			declare_war_with_cb = {
				casus_belli = cb_monster_vs_civ
				who = ROOT
			}
		}
		
		set_country_flag = F70_gnollakaz_strike

		kill_ruler = yes

		# Kaskanrit Irsmaset takes over as regent while the heir is underage.
		define_ruler = {
			name = "Kaskanrit Irsmaset"
			age = 35
			adm = 3
			dip = 2
			mil = 5
			#fixed = yes
			claim = 90
			culture = irsmahapan
			#regency = yes
		}
		add_ruler_personality = strict_personality

		define_ruler_to_general = {
			fire = 1
			shock = 5
			manuever = 2
			siege = 2
		}

		country_event = {
			id = flavour_mattabo.5
			days = 30
		}
	}
}
