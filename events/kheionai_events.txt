				set_variable = { deathwinds_low_level_var = 1 }
				set_variable = { deathwinds_months_var = 13 } #Time left until next Deathwind, 13 months
				set_variable = { deathwinds_active_months_var = -71 }
			}
			2728 = { #Arpedifer
				set_province_flag = deathwinds_warning_event
				province_event = {
					id = kheionai.2 #Warning event happens 6 months before
					days = 225 #June 1st 1445 (minus 6 days)
				}
			}
		}
	}

#Deathwinds Handler Event

#Deathwinds start > Notif event happens, add deathwind modifiers based on province group and windward level
# > 1 year later destruction event happens
# > 1 year later deathwinds end notif event happens (modifiers go away on their own)
# > 6.5 years later deathwinds warning event
# > 0.5 years later deathwinds start again
province_event = {
	id = kheionai.1
	title = none
	desc = none
	picture = none

	is_triggered_only = yes
	hidden = yes

	trigger = {
		has_global_flag = enable_deathwinds_handler_flag
		NOT = { has_global_flag = permanently_ended_deathwinds }
	}

	option = {
		name = none
		#Cycles between 4 events based one which province flag is set
		trigger_switch = {
			on_trigger = has_province_flag
			deathwinds_warning_event = {
				#Every country with a province in Alecand gets the deathwinds warning event
				alecand_region = { owner = { country_event = { id = kheionai.3 } } }

				#Prep the deathwinds start event
				province_event = { id = kheionai.2 days = 175 } #6 months (Couple days behind to ensure the next monthly pulse fires it)
				clr_province_flag = deathwinds_warning_event
				set_province_flag = deathwinds_start_event
			}
			deathwinds_start_event = {
				#Every country with a province in Alecand gets the deathwinds start event
				2936 = { setup_deathwinds = yes } #Metei
				alecand_region = { owner = { country_event = { id = kheionai.4 } } }

				#Prep the deathwinds destruction event
				province_event = { id = kheionai.2 days = 360 } #1 year
				clr_province_flag = deathwinds_start_event
				set_province_flag = deathwinds_destruction_event
			}
			deathwinds_destruction_event = {
				#Every country with a province in Alecand gets the deathwinds destruction event
				alecand_region = { owner = { country_event = { id = kheionai.5 } } }

				#Prep the deathwinds ends event
				province_event = { id = kheionai.2 days = 360 } #1 year
				clr_province_flag = deathwinds_destruction_event
				set_province_flag = deathwinds_end_event
			}
			deathwinds_end_event = {
				#Every country with a province in Alecand gets the deathwinds ends event
				end_deathwinds = yes
				alecand_region = { owner = { country_event = { id = kheionai.6 } } }

				#Prep the deathwinds warning event
				province_event = { id = kheionai.2 days = 2365 } #6.5 years
				clr_province_flag = deathwinds_end_event
				set_province_flag = deathwinds_warning_event
			}
		}
		clr_global_flag = enable_deathwinds_handler_flag
	}
}

#Deathwinds destruction event
country_event = {
	id = kheionai.5
	title = kheionai.5.t
	desc = kheionai.5.desc
	picture = LOOMING_DEATH_eventPicture
	
	is_triggered_only = yes
	
	#Flags so the event only fires once and not once per province in Alecand
	trigger = { NOT = { has_country_flag = deathwinds_disable_events_flag } NOT = { has_country_flag = deathwinds_devastation_event_flag } }
	after = { clr_country_flag = deathwinds_devastation_event_flag }

	immediate = {
		hidden_effect = {
			set_country_flag = deathwinds_devastation_event_flag
			every_owned_province = {
				limit = { has_deathwind_modifier = yes }
				#Every affected provicne gains 5 devastation per deathwind level in the provicne
				if = {
					limit = { NOT = { has_province_flag = windwards_reinforce_with_manpower_flag } }
					for_variable_amount = {
						variable = deathwinds_warded_var
						effect = "add_devastation = 5"
					}
				}
				else = { clr_province_flag = windwards_reinforce_with_manpower_flag }
				
				#If the deathwinds are at level 7+, destroy a random tax, manpower, forcelimit, dock, shipyard, or production building
				if = {
					limit = {
						OR = {
							has_province_modifier = deathwind_level_7
							has_province_modifier = deathwind_level_8
							has_province_modifier = deathwind_level_9
							has_province_modifier = deathwind_level_10
						}
					}
					random_list = {
						1 = {
							trigger = { has_tax_building_trigger = yes }
							remove_building = temple
							remove_building = cathedral
						}
						1 = {
							trigger = { has_manpower_building_trigger = yes }
							remove_building = barracks
							remove_building = training_fields
						}
						1 = {
							trigger = { has_forcelimit_building_trigger = yes }
							remove_building = regimental_camp
							remove_building = conscription_center
						}
						1 = {
							trigger = { has_dock_building_trigger = yes }
							remove_building = dock
							remove_building = drydock
						}
						1 = {
							trigger = { has_shipyard_building_trigger = yes }
							remove_building = shipyard
							remove_building = grand_shipyard
						}
						1 = {
							trigger = { has_production_building_trigger = yes }
							remove_building = workshop
							remove_building = counting_house
						}
					}
				}
			}
			#If deathwinds are 7+ in any province destroy 10 units in Alecand
			if = {
				limit = {
					any_owned_province = {
						OR = {
							has_province_modifier = deathwind_level_7
							has_province_modifier = deathwind_level_8
							has_province_modifier = deathwind_level_9
							has_province_modifier = deathwind_level_10
						}
					}
				}
				for = {
					amount = 10
					effect = "
						random_owned_province = {
							limit = {
								has_deathwind_modifier = yes
								land_units_in_province = { effect = ROOT }
							}
							kill_land_units = { who = ROOT amount = 1 }
						}
					"
				}
			}
			else_if = { #Otherwise if they're 4-6 in any province destroy 5 units in Alecand
				limit = {
					any_owned_province = {
						OR = {
							has_province_modifier = deathwind_level_4
							has_province_modifier = deathwind_level_5
							has_province_modifier = deathwind_level_6
						}
					}
				}
				for = {
					amount = 5
					effect = "
						random_owned_province = {
							limit = {
								has_deathwind_modifier = yes
								land_units_in_province = { effect = ROOT }
							}
							kill_land_units = { who = ROOT amount = 1 }
						}
					"
				}
			}
		}
	}

	option = {
		name = kheionai.5.a
		trigger = { any_owned_province = { has_deathwind_modifier = yes } }

		custom_tooltip = deathwinds_devastation_tt

		if = {
			limit = {
				any_owned_province = {
					OR = {
						has_province_modifier = deathwind_level_7
						has_province_modifier = deathwind_level_8
						has_province_modifier = deathwind_level_9
						has_province_modifier = deathwind_level_10
					}
				}
			}
			custom_tooltip = deathwinds_building_yes_tt
			custom_tooltip = deathwinds_units_yes_tt
			custom_tooltip = deathwinds_units_small_no_tt
		}
		else_if = {
			limit = {
				any_owned_province = {
					OR = {
						has_province_modifier = deathwind_level_4
						has_province_modifier = deathwind_level_5
						has_province_modifier = deathwind_level_6
					}
				}
			}
			custom_tooltip = deathwinds_building_no_tt
			custom_tooltip = deathwinds_units_no_tt
			custom_tooltip = deathwinds_units_small_yes_tt
		}
		else = {
			custom_tooltip = deathwinds_building_no_tt
			custom_tooltip = deathwinds_units_no_tt
			custom_tooltip = deathwinds_units_small_no_tt
		}
	}

	option = {
		name = kheionai.5.b
		highlight = yes
		trigger = {
			custom_trigger_tooltip = {
				tooltip = no_deathwinds_tt
				NOT = { any_owned_province = { has_deathwind_modifier = yes } }
			}
		}
		add_splendor = 10
	}

	option = {
		name = kheionai.3.b
		trigger = { has_not_full_deathwinds_system = yes }
		disable_deathwind_events = yes
	}
}

country_event = {
	id = kheionai.100
	title = kheionai.100.t
	desc = kheionai.100.d
	picture = EUROPEAN_REFUGEES_eventPicture
	
	trigger = {
		culture_group = kheionai_ruinborn_elf
		NOT = { tag = G52 } #Ameion
		NOT = { tag = G00 } #Nakyra
		capital_scope = {
			NOT = { region = alecand_region }
		}
		NOT = { has_country_flag = new_kheionai_culture }
		NOT = { has_country_flag = no_formable_culture_shift }
		if = {
			limit = { 
				tag = G56 #Deyeion
				ai = no
			}
			has_country_flag = G56_independence_flag
		}
		
		is_at_war = no
		OR = {
			is_subject = no
			has_country_flag = G56_independence_flag
		}
		
	}
	
	mean_time_to_happen = {
		months = 200
		modifier = {
			factor = 0.7
			stability = 1
		}
		modifier = {
			factor = 0.25
			num_of_cities = 10
