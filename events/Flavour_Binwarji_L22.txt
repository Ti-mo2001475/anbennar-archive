province_event = { #Province Ask for Investment
	id = binwarji.102
	title = binwarji.102.t
	desc = binwarji.102.d
	picture = NEGUSA_NAGAST_eventPicture

	is_triggered_only = yes

	option = {
		name = binwarji.102.a

		owner = {
			if = {
				limit = { ruler_has_personality = architectural_visionary_personality }
				add_years_of_income = -0.025
				custom_tooltip = L22_architect_discount_TT
			}
			else = {
				add_years_of_income = -0.05
			}
			add_adm_power = -50
		}
		add_devastation = -100
		random_list = {
			33 = { add_base_tax = 1 }
			33 = { add_base_production = 1 }
			33 = { add_base_manpower = 1 }
		}
		add_prosperity = 25
		area = {
			limit = {
				country_or_non_sovereign_subject_holds = ROOT
			}
			add_province_modifier = {
				name = L22_court_helped_city
				duration = 1825
			}
		}
		L22_court_of_fangaula_helped_locals_effect = yes
	}

	option = {
		name = binwarji.102.b

		add_prestige = -5
	}
}

province_event = { #Priest Wanst Investment
	id = binwarji.104
	title = binwarji.104.t
	desc = binwarji.104.d
	picture = NEGUSA_NAGAST_eventPicture

	is_triggered_only = yes

	option = {
		name = binwarji.104.a

		owner = { add_adm_power = -50 }

		area = {
			limit = {
				country_or_non_sovereign_subject_holds = ROOT
			}
			random_owned_or_subject_province = {
				if = {
					limit = { 
						OR = {
							has_tax_building_trigger = yes
							has_construction = temple
							has_construction = cathedral
						}
					}
					if = {
						limit = { owner = { ruler_has_personality = zealot_personality } }
						add_base_tax = 2
						custom_tooltip = L22_zealot_discount_TT
					}
					else = {
						add_base_tax = 1
					}
				}
				else = {
					add_building = temple
				}
			}
			add_province_modifier = {
				name = L22_court_helped_city
				duration = 1825
			}
		}
		L22_court_of_fangaula_helped_locals_effect = yes
	}

	option = {
		name = binwarji.104.b
