end_random_favor = {
	custom_tooltip = end_random_$amount$_favor_tt #up to 5 random favors have loc
	hidden_effect = {
		random_list = {
			1 = {
				change_variable = {
					which = end_noble_candidate_favor
					value = $amount$
				}
			}
			1 = {
				change_variable = {
					which = end_local_candidate_favor
					value = $amount$
				}
			}
			1 = {
				trigger = { has_country_flag = end_elven_unlocked }
				change_variable = {
					which = end_elven_candidate_favor
					value = $amount$
				}
			}
			1 = {
				trigger = {	has_country_flag = end_elven_unlocked }
				change_variable = {
					which = end_mistmancer_candidate_favor
					value = $amount$
				}
			}
			1 = {
				trigger = {	has_country_flag = end_elven_unlocked }
				change_variable = {
					which = end_agrarian_candidate_favor
					value = $amount$
				}
			}
			1 = {
				trigger = {	has_country_flag = end_elven_unlocked }
				change_variable = {
					which = end_industrialist_candidate_favor
					value = $amount$
				}
			}
			1 = {
				trigger = {	has_country_flag = end_elven_unlocked }
				change_variable = {
					which = end_imperialist_candidate_favor
					value = $amount$
				}
			}
		}
	}
}

end_add_favor_to_current = {
	custom_tooltip = end_current_ruling_$amount$_favor_tt #up to 5 has loc
	hidden_effect = {
		if = {
			limit = { has_ruler_flag = end_noble_candidate_ruler }
			change_variable = {
				which = end_noble_candidate_favor
				value = $amount$
			}
		}
		else_if = {
			limit = { has_ruler_flag = end_local_candidate_ruler }
			change_variable = {
				which = end_local_candidate_favor
				value = $amount$
			}
		}
		else_if = {
			limit = { has_ruler_flag = end_elven_candidate_ruler }
			change_variable = {
				which = end_elven_candidate_favor
				value = $amount$
			}
		}
		else_if = {
			limit = { has_ruler_flag = end_mistmancer_candidate_ruler }
			change_variable = {
				which = end_mistmancer_candidate_favor
				value = $amount$
			}
		}
		else_if = {
			limit = { has_ruler_flag = end_agrarian_candidate_ruler }
			change_variable = {
				which = end_agrarian_candidate_favor
				value = $amount$
			}
		}
		else_if = {
			limit = { has_ruler_flag = end_industrialist_candidate_ruler }
			change_variable = {
				which = end_industrialist_candidate_favor
				value = $amount$
			}
		}
		else = {
			change_variable = {
				which = end_imperialist_candidate_favor
				value = $amount$
			}
		}
	}
}

end_increase_factionalism = {
	if = {
		limit = { has_country_modifier = end_factionalism_5 }
		hidden_effect = { remove_country_modifier = end_factionalism_5 }
		add_country_modifier = {
			name = end_factionalism_6
			duration = -1
		}
	}
	else_if = {
		limit = { has_country_modifier = end_factionalism_4 }
		hidden_effect = { remove_country_modifier = end_factionalism_4 }
		add_country_modifier = {
			name = end_factionalism_5
			duration = -1
		}
	}
	else_if = {
		limit = { has_country_modifier = end_factionalism_3 }
		hidden_effect = { remove_country_modifier = end_factionalism_3 }
		add_country_modifier = {
			name = end_factionalism_4
			duration = -1
		}
	}
	else_if = {
		limit = { has_country_modifier = end_factionalism_2 }
		hidden_effect = { remove_country_modifier = end_factionalism_2 }
		add_country_modifier = {
			name = end_factionalism_3
			duration = -1
		}
	}
	else_if = {
		limit = { has_country_modifier = end_factionalism_1 }
		hidden_effect = { remove_country_modifier = end_factionalism_1 }
		add_country_modifier = {
			name = end_factionalism_2
			duration = -1
		}
	}
	else = {
		add_country_modifier = {
			name = end_factionalism_1
			duration = -1
		}
	}
}
