#Add the appropriate deathwinds modifiers to the region
calculate_current_deathwind_intensity = {
	#Get the variables range
	set_variable = { deathwinds_rand_var = $high_level$ }
	change_variable = { deathwinds_rand_var = -$low_level$ }
	#Set it to a random number in that range
	change_variable = { deathwinds_rand_var = 1 } #Highest value is excluded in random_variable, so this way the max is [high_value].999
	random_variable = { which = deathwinds_var which = deathwinds_rand_var }
	#Remove any decimals
	round_variable = { which = deathwinds_var value = -1 }
	#Ensure it's at or above the lowest allowed value
	change_variable = { deathwinds_var = $low_level$ }
	#If the Deathwinds have been increased in a seperate effect, increase it by that amount
	change_variable = { which = deathwinds_var which = deathwinds_increase_var }
	if = { #Deathwinds can't go about 10
		limit = { check_variable = { deathwinds_var = 11 } }
		set_variable = { deathwinds_var = 10 }
	}
	else_if = { #Deathwinds can't go below 0
		limit = { NOT = { check_variable = { deathwinds_var = 1 } } }
		set_variable = { deathwinds_var = 0 }
	}
	#Change to a different modifier so the other one can be used to display the unwarded deathwinds level
	set_variable = { which = deathwinds_warded_var which = deathwinds_var } 
}

#Sets variables that say the possible low and high level that the next Deathwind can be. Used in aPUI_deathwinds_bar_tt
set_next_deathwind_intensity = {
	set_variable = { deathwinds_low_level_var = $low_level$ }
	set_variable = { deathwinds_high_level_var = $high_level$ }

	#If the Deathwinds have been increased in a seperate effect, increase it by that amount
	change_variable = { which = deathwinds_low_level_var which = deathwinds_increase_var }
	change_variable = { which = deathwinds_high_level_var which = deathwinds_increase_var }
	if = { #Deathwinds can't go about 10
		limit = { check_variable = { deathwinds_high_level_var = 11 } }
		set_variable = { deathwinds_high_level_var = 10 }
		if = {
			limit = { check_variable = { deathwinds_low_level_var = 11 } }
			set_variable = { deathwinds_low_level_var = 10 }
		}
	}
	if = { #Deathwinds can't go below 0
		limit = { NOT = { check_variable = { deathwinds_low_level_var = 1 } } }
		set_variable = { deathwinds_low_level_var = 0 }
		if = {
			limit = { NOT = { check_variable = { deathwinds_high_level_var = 1 } } }
			set_variable = { deathwinds_high_level_var = 0 }
		}
	}
}

apply_current_deathwind_intensity = {
	alecand_region = {
		set_variable = { which = deathwinds_warded_var which = PREV }
		#Change it depending on where in the region the province is in
		trigger_switch = {
			on_trigger = province_group
			alecandi_capital_provinces = { change_variable = { deathwinds_warded_var = -10 } }
			deathwinds_minus1_group = { change_variable = { deathwinds_warded_var = -1 } }
			deathwinds_minus2_group = { change_variable = { deathwinds_warded_var = -2 } }
			deathwinds_minus3_group = { change_variable = { deathwinds_warded_var = -3 } }
			deathwinds_plus1_group = { change_variable = { deathwinds_warded_var = 1 } }
			deathwinds_plus3_group = { change_variable = { deathwinds_warded_var = 3 } }
		}

		#Reduce it by the provinces windward level
		trigger_switch = {
			on_trigger = has_province_modifier
			windward_level_10 = { change_variable = { deathwinds_warded_var = -10 } }
			windward_level_9 = { change_variable = { deathwinds_warded_var = -9 } }
			windward_level_8 = { change_variable = { deathwinds_warded_var = -8 } }
			windward_level_7 = { change_variable = { deathwinds_warded_var = -7 } }
			windward_level_6 = { change_variable = { deathwinds_warded_var = -6 } }
			windward_level_5 = { change_variable = { deathwinds_warded_var = -5 } }
			windward_level_4 = { change_variable = { deathwinds_warded_var = -4 } }
			windward_level_3 = { change_variable = { deathwinds_warded_var = -3 } }
			windward_level_2 = { change_variable = { deathwinds_warded_var = -2 } }
			windward_level_1 = { change_variable = { deathwinds_warded_var = -1 } }
		}

		#kheion gov reform
		if = {
			limit = { owner = { has_reform = tame_the_storm } }
			if = {
				limit = { has_province_modifier = abjuration_1_ward_mod_3 }
				change_variable = { deathwinds_warded_var = -4 }
			}
			else_if = {
				limit = { has_province_modifier = abjuration_1_ward_mod_2 }
				change_variable = { deathwinds_warded_var = -3 }
			}
			else_if = {
				limit = { has_province_modifier = abjuration_1_ward_mod_1 }
				change_variable = { deathwinds_warded_var = -2 }
			}
			else_if = {
				limit = { has_province_modifier = abjuration_1_ward_mod_0 }
				change_variable = { deathwinds_warded_var = -1 }
			}

			if = {
				limit = { has_building = mage_tower }
				change_variable = { deathwinds_warded_var = -1 }
			}
		}
		
		#Assign the appropriate modifier
		assign_deathwinds_modifier = yes

		#Reset the aPUI bar to 0
		set_aPUI_var_bar_provincial = { name = deathwinds amount = 0 }
	}
}

#Effect to assign the appropriate deathwinds modifier based on the provinces current deathwinds level after warding
 #A seperate scripted effect so the 'Reinforce with Mana' button can use it too

	else_if = { limit = { check_variable = { deathwinds_warded_var = 9 } } add_permanent_province_modifier = { name = deathwind_level_9 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 8 } } add_permanent_province_modifier = { name = deathwind_level_8 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 7 } } add_permanent_province_modifier = { name = deathwind_level_7 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 6 } } add_permanent_province_modifier = { name = deathwind_level_6 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 5 } } add_permanent_province_modifier = { name = deathwind_level_5 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 4 } } add_permanent_province_modifier = { name = deathwind_level_4 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 3 } } add_permanent_province_modifier = { name = deathwind_level_3 duration = 730 } }
	else_if = { limit = { check_variable = { deathwinds_warded_var = 2 } } add_permanent_province_modifier = { name = deathwind_level_2 duration = 730 } }

	else = { set_variable = { deathwinds_warded_var = 0 } } #If it's below 0, set it to 0 for the UI tooltip

#Ends the deathwinds for the period
end_deathwinds = {
	clr_global_flag = deathwinds_active_flag
	#The UI tooltip displays '-1' as the deathwinds not being active
	2936 = { #Metei
		set_variable = { which = deathwinds_previous_var which = deathwinds_var } #To display the last Deathwind level
		set_variable = { deathwinds_var = -1 }
	}
	alecand_region = {
		remove_all_deathwinds_modifiers = yes
		set_variable = { deathwinds_warded_var = -1 }
	}
}

#Completely disable the Deathwinds system, no more deathwinds, warding, aPUI, etc.
permanently_end_deathwinds = {
	end_deathwinds = yes
	set_global_flag = permanently_ended_deathwinds
	alecand_region = {
		remove_all_windward_modifiers = yes
	}
}

#Update the variables that show the bar progress in the aPUI and how many months are left before it starts or ends
update_deathwinds_bar = {
	2936 = { #Metei
		change_aPUI_var_bar_provincial = { name = deathwinds amount = 1 } #Tracks the value of the deathwinds aPUI bar

		set_variable = { deathwinds_months_var = 108 } #Set variable to 9 years in months (total Deathwinds cycle time)
		set_variable = { deathwinds_active_months_var = 24 } #Set variable to 2 years in months (total time Deathwinds are active)

		subtract_variable = { #Subtract it by the current month to get the months left in total cycle
			which = deathwinds_months_var
			which = aPUI_deathwinds_var
		}
		subtract_variable = { #Subtract it by the current month to get the months left while deathwinds are active
			which = deathwinds_active_months_var
			which = aPUI_deathwinds_var
		}
	
		if = { #If a player has the deathwinds menu open, update the aPUI bar
			limit = { has_global_flag = player_has_deathwinds_apui_open }
			alecand_region = { set_aPUI_var_bar_provincial = { name = deathwinds var_amount = PREV } }
		}
	}
}
