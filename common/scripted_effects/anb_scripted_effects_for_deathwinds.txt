# #Setup deathwinds according to the rules for the given region, calls a second scripted effect with a random list, currently only takes alecand = yes as argument, can be expanded to other regions
setup_deathwinds = {
	if = {
		limit = { is_year = 1700 }
		get_deathwinds_level_for_region = {
			low_level = 10
			mid_level = 10
			high_level = 10
		}
	}
	else_if = {
		limit = { is_year = 1666 }
		get_deathwinds_level_for_region = {
			low_level = 9
			mid_level = 10
			high_level = 10
		}
	}
	else_if = {
		limit = { is_year = 1633 }
		get_deathwinds_level_for_region = {
			low_level = 8
			mid_level = 9
			high_level = 10
		}
	}
	else_if = {
		limit = { is_year = 1600 }
		get_deathwinds_level_for_region = {
			low_level = 7
			mid_level = 8
			high_level = 9
		}
	}
	else_if = {
		limit = { is_year = 1566 }
		get_deathwinds_level_for_region = {
			low_level = 6
			mid_level = 7
			high_level = 8
		}
	}
	else_if = {
		limit = { is_year = 1533 }
		get_deathwinds_level_for_region = {
			low_level = 5
			mid_level = 6
			high_level = 7
		}
	}
	else_if = {
		limit = { is_year = 1500 }
		get_deathwinds_level_for_region = {
			low_level = 4
			mid_level = 5
			high_level = 6
		}
	}
	else_if = {
		limit = { is_year = 1466 }
		get_deathwinds_level_for_region = {
			low_level = 3
			mid_level = 4
			high_level = 5
		}
	}
	else_if = {
		limit = { is_year = 1450 }
		get_deathwinds_level_for_region = {
			low_level = 2
			mid_level = 3
			high_level = 4
		}
	}
	#An easy one to start with
	else = {
		get_deathwinds_level_for_region = {
			low_level = 1
			mid_level = 1
			high_level = 1
		}
	}
}

#Deathwinds start > Notif event happens, add deathwind modifiers based on province group and windward level
# > 1 year later destruction event happens
# > 1 year later deathwinds end notif event happens (modifiers go away on their own)
# > 7 years later deathwinds start again

# #Find the deathwinds level and checks it against each province in the region, take low_level, mid_level and high_level as level, and affected_region as region
get_deathwinds_level_for_region = {
	random_list = {
		1 = {
			alecand_region = {
				add_permanent_province_modifier = {
					name = deathwind_level_$low_level$
					duration = 730 #2 years
				}
			}
		}
		1 = {
			alecand_region = {
				add_permanent_province_modifier = {
					name = deathwind_level_$mid_level$
					duration = 730 #2 years
				}
			}
		}
		1 = {
			alecand_region = {
				add_permanent_province_modifier = {
					name = deathwind_level_$high_level$
					duration = 730 #2 years
				}
			}
		}
	}
}
