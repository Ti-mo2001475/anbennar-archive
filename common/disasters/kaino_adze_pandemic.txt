
kaino_adze_pandemic = {
	potential = {
		has_country_flag = kaino_can_begin_adze_pandemic
	}

	
	
	can_start = {
		custom_trigger_tooltip = {
			tooltip = kaino_can_start_adze_pandemic_tt
			always = yes
		}
	}
	
	can_stop = {
		custom_trigger_tooltip = {
			tooltip = kaino_can_stop_adze_pandemic_tt
			has_country_flag = kaino_flag_cured_adze_poisoning
		}
	}

	progress = {
		modifier = {
			has_country_modifier = kaino_adze_poisoning
			factor = 0.01
		}
		modifier = {
			has_country_modifier = kaino_curious_disease
			factor = 0.01
		}
		modifier = { 
			has_country_modifier = kaino_carrion_call_adze
			factor = 2
		}
	}

	modifier = {
		manpower_recovery_speed = -0.9
		morale_damage_received = 0.2
		stability_cost_modifier = 2.5
	}
	
	can_end = {
		custom_trigger_tooltip = {
			tooltip = kaino_can_stop_adze_pandemic_tt
			has_country_flag = kaino_flag_cured_adze_poisoning
		}
	}
	on_start = flavour_kaino.100
	on_end = flavour_kaino.101
}
