kalavendy_gos_ref = {  #Идея в том что калавендские лоры
	icon = "kalavendy_gos_ref"
	allow_normal_conversion = yes
	potential = {
		culture = kalavendhi
	}
	trigger = {
		culture = kalavendhi
	}
	modifiers = {
		manpower_recovery_speed = 0.1
		global_prosperity_growth = 0.02
	}
	effect = {
		custom_tooltip = kalavendy_gos_ref_tt
		hidden_effect = {
			every_owned_province = {
				if = {
					limit = { province_with_arable_goods = yes region = kalavend_region }
					add_province_modifier = {
						name = kalavendy_population #+15 local manpower and 20% trade value modifier
						duration = -1
						hidden = yes
					}
				}
			}
		}
	}
	removed_effect = {
		hidden_effect = {
			every_owned_province = {
				limit = { has_province_modifier = kalavendy_population }
				remove_province_modifier = kalavendy_population
			}
		}
	}
}
