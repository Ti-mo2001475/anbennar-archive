    icon = "giga_chad"
	allow_normal_conversion = yes
	potential = {
		culture_group = kheionai_ruinborn_elf
		capital_scope = { 
			region = alecand_region
		}
	}
	trigger = {
		culture_group = kheionai_ruinborn_elf
		capital_scope = { 
			region = alecand_region
		}
	}
	conditional = {
		allow = {
			has_dlc = "Res Publica"
		}
		can_use_trade_post = yes
	}
	conditional = {
		allow = {
			has_dlc = "Domination"
		}
		custom_attributes = {
			can_establish_trade_protectorates = yes
		}
	}
	boost_income = yes
	modifiers = {
		governing_capacity_modifier = -0.33
		liberty_desire_from_subject_development = -0.25
	}
	effect = {
		custom_tooltip = xechtari_above_all_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					region = alecand_region
				}
				add_province_triggered_modifier = xechtari_above_all_ptm
			}
		}
		removed_effect = {
			remove_province_triggered_modifier = abjuration_spell_three_quarter
		}
	}
}
