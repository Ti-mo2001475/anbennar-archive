is_valid_nekhei = {
	if = {
		limit = { overlord = { OR = { tag = G51 was_tag = G51 } } } #Arpedifer
		tag = G53 #Empkeios
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G47 was_tag = G47 } } } #Oktikheion
		tag = G58 #Keyolion
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G49 was_tag = G49 } } } #Lokemeion
		tag = G54 #Besolaki
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G50 was_tag = G50 } } } #Ormam
		tag = G57 #Apikhoxi
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G48 was_tag = G48 } } } #Kherka
		tag = G59 #Anisikheion
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G61 was_tag = G61 } } } #Kimanis
		tag = G62 #Amgremos
		is_subject_of_type = nekhei
	}
	else_if = {
		limit = { overlord = { OR = { tag = G46 was_tag = G46 } } } #Degakheion
		is_subject_of_type = nekhei
		OR = { 
			tag = G55 #Eneion
			tag = G56 #Deyeion
			AND = {
				tag = G62 #Amgremos
				overlord = { has_country_flag = G46_amgremos_nekhei }
			}
		}
	}
	else = { always = no }
}

G54_faction_subtrigger = {
	if = {
		limit = { faction_in_power = $faction$ }
		$trigger$ = {
			who = $tag$
			value = $low_value$
		}
	}
	else = {
		$trigger$ = {
			who = $tag$
			value = $high_value$
		}
	}
}
G54_faction_trigger = {
	if = {
		limit = { exists = $tag$ }
		OR = {
			G54_faction_subtrigger = {
				faction = beso_skighubhen
				trigger = has_spy_network_in
				tag = $tag$
				low_value = 75
				high_value = 100
			}
			AND = {
				G54_faction_subtrigger = {
					faction = beso_nautabhen
					trigger = reverse_has_opinion
					tag = $tag$
					low_value = 75
					high_value = 100
				}
				G54_faction_subtrigger = {
					faction = beso_skighubhen
					trigger = has_spy_network_in
					tag = $tag$
					low_value = 25
					high_value = 50
				}
			}
			custom_trigger_tooltip = {
				tooltip = G54_faction_trigger_tt
				always = no
			}
		}
	}
	else = {
		[[$area$]
			$area$ = {
				type = all
				country_or_non_sovereign_subject_holds = ROOT
			}
		]
	}
}

G54_faction_highlight = {
	OR = {
		AND = {
			exists = $tag$
			is_capital_of = $tag$
		}
		[[area]
			AND = {
				NOT = { exists = $tag$ }
				area = $area$
				NOT = { country_or_non_sovereign_subject_holds = ROOT }
			}
		]
	}
	ROOT = {
		NOT = {
			AND = {
				[[additional_trigger] $additional_trigger$ ]
				G54_faction_trigger = {
					tag = $tag$
					[[area] area = $area$ ]
				}
			}
		}
	}
}

#Trigger for the 2nd Deathwinds aPUI button
windwards_reinforce_with_manpower_trigger = {
	if = {
		limit = { NOT = { owned_by = FROM } }
		owned_by = FROM
	}
	has_full_deathwinds_system_button = yes
	#Requires manpower equal to 5 + (2 * Maximum Potential Deathwind Level) 
	if = {
		limit = { is_year = 1520 } #Level 10 Max = 5+20=25
		custom_trigger_tooltip = {
			tooltip = owner_has_manpower_25_tt
			owner = { manpower = 25 }
		}
	}
	if = {
		limit = { is_year = 1480 NOT = { is_year = 1520 } } #Level 8 Max = 5+16=21
		custom_trigger_tooltip = {
			tooltip = owner_has_manpower_21_tt
			owner = { manpower = 21 }
		}
	}
	if = {
		limit = { is_year = 1450 NOT = { is_year = 1480 } } #Level 4 Max = 5+8=13
		custom_trigger_tooltip = {
			tooltip = owner_has_manpower_13_tt
			owner = { manpower = 13 }
		}
	}
	if = { #Level 1 Max
		limit = { NOT = { is_year = 1450 } } #Level 4 Max = 5+2=7
		custom_trigger_tooltip = {
			tooltip = owner_has_manpower_7_tt
			owner = { manpower = 7 }
		}
	}
	custom_trigger_tooltip = {
		tooltip = windwards_reinforce_with_manpower_trigger_tt
		NOT = { has_province_flag = windwards_reinforce_with_manpower_flag }
	}
}
