edict_religious_unity = {
	potential = {
		always = yes		#we support "potential" if modders want to have lots and just show some.
	}
	
	allow = {
		always = yes
	}
	
	modifier = {
		local_missionary_strength = 0.01
	}
	
	#ROOT - Country, FROM - Province, if all provinces in a state, that are owned by the country,
	#	meet this requirement, the alert will appear.
	
	notify_trigger = {
		FROM = {
			has_owner_religion = yes
		}
	}
	
	color = { 117 38 143 }
	
	ai_will_do = {
		factor = 100
		modifier = {
			factor = 0
			all_province_in_state = {
				has_missionary = no
			}
		}
		modifier = {
			factor = 3
			any_province_in_state = {
				has_missionary = yes
			}
		}
	}
	on_state_edict_enacted = {
		if = {
			limit = { FROM = { mission_completed = F70_restructure_the_priests } }
			add_province_modifier = {
				name = F70_restructure_priests
				duration = -1
			}
		}
	 }
	on_state_edict_revoked = {
		remove_province_modifier = F70_restructured_priests
		if = {
			limit = { FROM = { mission_completed = F70_restructure_the_priests } }
			FROM = { F70_update_church_power_modifier = yes }
		}
	}
}
